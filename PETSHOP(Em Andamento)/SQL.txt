spring.jpa.hibernate.ddl-auto=create

# Para:
spring.jpa.hibernate.ddl-auto=update

    function applyFilters() {
        let filtered = [...allProducts];
        
        const minPrice = parseFloat(document.getElementById('priceMin').value) || 0;
        const maxPrice = parseFloat(document.getElementById('priceMax').value) || Infinity;
        filtered = filtered.filter(p => {
            const price = p.precoDesconto || 0;
            return price >= minPrice && price <= maxPrice;
        });
        
        if (document.getElementById('promoOnly').checked) {
            filtered = filtered.filter(p => p.precoDesconto && p.preco && p.precoDesconto < p.preco);
        }
        
        if (document.getElementById('inStockOnly').checked) {
            filtered = filtered.filter(p => p.qtdEstoque > 0);
        }
        
        const sortValue = document.getElementById('sortSelect').value;
        switch(sortValue) {
            case 'price-asc': filtered.sort((a, b) => (a.precoDesconto || 0) - (b.precoDesconto || 0)); break;
            case 'price-desc': filtered.sort((a, b) => (b.precoDesconto || 0) - (a.precoDesconto || 0)); break;
            case 'name-asc': filtered.sort((a, b) => a.nome.localeCompare(b.nome)); break;
            case 'name-desc': filtered.sort((a, b) => b.nome.localeCompare(a.nome)); break;
            case 'stock-desc': filtered.sort((a, b) => (b.qtdEstoque || 0) - (a.qtdEstoque || 0)); break;
        }
        
        renderProducts(filtered);
        document.getElementById('productGrid').scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

    function clearFilters() {
        document.getElementById('sortSelect').value = 'default';
        document.getElementById('priceMin').value = '';
        document.getElementById('priceMax').value = '';
        document.getElementById('promoOnly').checked = false;
        document.getElementById('inStockOnly').checked = false;
        renderProducts(allProducts);
    }

    document.getElementById('categoryForm').addEventListener('submit', async (e) => {
        e.preventDefault();
        clearErrors();
        let valid = true;
        
        if (!document.getElementById('categoryName').value.trim()) {
            showFieldError('categoryName', 'categoryNameError');
            valid = false;
        }
        if (!valid) return;
        
        showLoading('submitBtnText', 'submitBtnSpinner');
        const id = document.getElementById('categoryId').value;
        const categoria = {
            nome: document.getElementById('categoryName').value,
            descricao: document.getElementById('categoryDesc').value,
            icone: document.getElementById('categoryIcon').value,
            ativo: true
        };
        const method = id ? 'PUT' : 'POST';
        const url = id ? `${API_URL}/${id}` : API_URL;
        if(id) categoria.idCategoria = parseInt(id);

        try {
            const response = await fetch(url, {
                method: method,
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(categoria)
            });
            if (response.ok) { 
                resetForm(); 
                fetchCategories(); 
                showToast('Categoria salva com sucesso! ✅');
            }
        } catch (err) { 
            console.error(err); 
            showToast('Erro ao salvar categoria ❌', 'error');
        } finally {
            hideLoading('submitBtnText', 'submitBtnSpinner');
        }
    });

    document.getElementById('productForm').addEventListener('submit', async (e) => {
        e.preventDefault();
        clearErrors();
        let valid = true;
        
        if (!document.getElementById('productName').value.trim()) { showFieldError('productName', 'productNameError'); valid = false; }
        if (!document.getElementById('productCategory').value) { showFieldError('productCategory', 'productCategoryError'); valid = false; }
        if (!document.getElementById('productStock').value) { showFieldError('productStock', 'productStockError'); valid = false; }
        if (!document.getElementById('productPrice').value) { showFieldError('productPrice', 'productPriceError'); valid = false; }
        if (!document.getElementById('productDiscountPrice').value) { showFieldError('productDiscountPrice', 'productDiscountError'); valid = false; }
        if (!valid) return;
        
        showLoading('productSubmitBtnText', 'productSubmitBtnSpinner');
        const id = document.getElementById('productId').value;
        const product = {
            nome: document.getElementById('productName').value,
            preco: parseFloat(document.getElementById('productPrice').value),
            precoDesconto: parseFloat(document.getElementById('productDiscountPrice').value),
            qtdEstoque: parseInt(document.getElementById('productStock').value),
            imagem: document.getElementById('productImage').value,
            ativo: true,
            categoria: { idCategoria: parseInt(document.getElementById('productCategory').value) }

        const url = id ? `${API_URL_PRODUTOS}/${id}` : API_URL_PRODUTOS;
        if(id) product.idProduto = parseInt(id);

        try {

    <script>
        // ===== INICIALIZAÇÃO DO TEMA =====
        function initTheme() {
            // Carregar tema
            const savedTheme = localStorage.getItem('petshop-theme');
            if (savedTheme === 'light') {
                document.body.classList.add('light-mode');
            }
            
            // Carregar cor
            const savedAccent = localStorage.getItem('petshop-accent');
            if (savedAccent) {
                document.documentElement.style.setProperty('--accent-rgb', savedAccent);
            }
            
            updateThemeIcon();
        }
        
        function setAccent(rgb) {
            document.documentElement.style.setProperty('--accent-rgb', rgb);
            localStorage.setItem('petshop-accent', rgb);
            lucide.createIcons();
        }
        
        function toggleTheme() {
            document.body.classList.toggle('light-mode');
            const isLight = document.body.classList.contains('light-mode');
            localStorage.setItem('petshop-theme', isLight ? 'light' : 'dark');
            updateThemeIcon();
        }
        
        function updateThemeIcon() {
            const icon = document.getElementById('themeIcon');
            if (icon) {
                const isLight = document.body.classList.contains('light-mode');
                icon.setAttribute('data-lucide', isLight ? 'moon' : 'sun');
                lucide.createIcons();
            }
        }
        
        function showToast(message, type = 'success') {
            const toast = document.createElement('div');
            toast.className = `toast fixed bottom-5 right-5 px-6 py-3 rounded-xl text-white text-sm font-bold z-[200] ${
                type === 'success' ? 'bg-emerald-500' : 'bg-red-500'
            }`;
            toast.textContent = message;
            document.body.appendChild(toast);
            setTimeout(() => toast.remove(), 3000);
        }
        
        // Inicializar tema imediatamente
        initTheme();
        lucide.createIcons();
        
        // ===== CARREGAR PRODUTO =====
        const urlParams = new URLSearchParams(window.location.search);
        const productId = urlParams.get('id');

        if (!productId) {
            document.getElementById('productDetail').innerHTML = `
                <div class="glass-card p-10 text-center">
                    <i data-lucide="alert-circle" class="w-16 h-16 text-red-400 mx-auto mb-4"></i>
                    <h2 class="text-2xl font-bold text-red-400 mb-2">Produto não especificado</h2>
                    <p class="text-gray-400">Nenhum ID de produto foi fornecido na URL.</p>
                    <a href="index.html" class="inline-block mt-4 bg-accent text-white px-6 py-2 rounded-xl text-sm font-bold hover:opacity-90 transition-all">Voltar à loja</a>
                </div>`;
            lucide.createIcons();
        } else {
            fetch(`http://localhost:8080/api/produtos/${productId}`)
                .then(response => {
                    if (!response.ok) throw new Error('Produto não encontrado');
                    return response.json();
                })
                .then(product => {
                    updateMetaTags(product);
                    renderProductDetail(product);
                })
                .catch(error => {
                    document.getElementById('productDetail').innerHTML = `
                        <div class="glass-card p-10 text-center">
                            <i data-lucide="frown" class="w-16 h-16 text-red-400 mx-auto mb-4"></i>
                            <h2 class="text-2xl font-bold text-red-400 mb-2">Erro</h2>
                            <p class="text-gray-400">${error.message}</p>
                            <a href="index.html" class="inline-block mt-4 bg-accent text-white px-6 py-2 rounded-xl text-sm font-bold hover:opacity-90 transition-all">Voltar à loja</a>
                        </div>`;
                    lucide.createIcons();
                });
        }

        function updateMetaTags(product) {
            const price = parseFloat(product.precoDesconto || 0).toFixed(2);
            document.title = `${product.nome} - TG Petshop`;
            
            const metaDesc = document.querySelector('meta[name="description"]');
            if (metaDesc) metaDesc.setAttribute('content', `${product.nome} - R$ ${price} | ${product.categoria ? product.categoria.nome : 'Premium'} | TG Petshop`);
            
            const ogTitle = document.querySelector('meta[property="og:title"]');
            if (ogTitle) ogTitle.setAttribute('content', `${product.nome} - TG Petshop`);
            
            const ogDesc = document.querySelector('meta[property="og:description"]');
            if (ogDesc) ogDesc.setAttribute('content', `${product.nome} por apenas R$ ${price}. Aproveite!`);
            
            const ogImage = document.querySelector('meta[property="og:image"]');
            if (ogImage) ogImage.setAttribute('content', product.imagem || 'https://images.unsplash.com/photo-1583511655857-d19b40a7a54e?w=600');
        }
